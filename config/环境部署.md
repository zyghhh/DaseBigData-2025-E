MYSQL（表以实验具体设计为准）

```plain
进入mysql命令行：sudo mysql

-- 1. 创建实验数据库
CREATE DATABASE stream_experiment;

-- 2. 切换到该库
USE stream_experiment;

-- 3. 创建指标表 (根据你的实验图设计)
-- 这张表用来存放 Metrics Collector 计算后的结果
--  如果存在旧表，先删除
DROP TABLE IF EXISTS metrics;

--  创建新表
CREATE TABLE metrics (
    -- [联合主键] 区分不同任务和不同消息
    job_type VARCHAR(10) NOT NULL,    -- 'flink' 或 'storm'
    msg_id BIGINT NOT NULL,           -- 消息的原始 ID (Generator 生成)

    -- [时间指标]
    event_time BIGINT,                -- 数据产生时间 (Generator 生成)
    in_time BIGINT,                   -- (可选) 进入计算任务的时间，如果你的 JSON 没传这个，可以存 0
    out_time BIGINT,                  -- 数据到达 Collector 的时间 (当前系统时间)
    
    -- [性能指标]
    latency BIGINT,                   -- 端到端延迟 (out_time - event_time)
    process_time BIGINT,              -- (可选) 处理耗时
    
    -- [★ 实验核心指标 ★]
    -- 记录该消息被处理的次数。
    -- 1 = 正常 (Exactly Once 效果)
    -- >1 = 发生重发 (At Least Once 效果)
    process_count INT DEFAULT 1,      

    -- [元数据]
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- 首次插入时间
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, -- 最后一次更新时间

    -- 设置联合主键，确保同一个任务的同一个 msg_id 只有一行记录
    PRIMARY KEY (job_type, msg_id)
);

-- 3. 创建索引 (可选，用于加速分析查询)
CREATE INDEX idx_process_count ON metrics(process_count);
CREATE INDEX idx_latency ON metrics(latency);

-- 4. 创建远程连接用户
-- 用户名: exp_user
-- 密码: password (你可以自己改复杂点，但实验环境简单就好)
-- '%' 表示允许从任何 IP 连接
CREATE USER 'exp_user'@'%' IDENTIFIED BY 'password';

-- 5. 授权
GRANT ALL PRIVILEGES ON stream_experiment.* TO 'exp_user'@'%';

-- 6. 刷新权限
FLUSH PRIVILEGES;

-- 7. 退出
EXIT;

node2,3连接测试：
# 尝试连接
mysql -h node1 -u exp_user -p
# 输入密码: password
```

## <font style="color:rgb(26, 28, 30);">Flink 1.14.6 分布式集群 (Standalone Mode) </font>
+ **<font style="color:rgb(26, 28, 30);">Node 1 (Master):</font>**<font style="color:rgb(26, 28, 30);"> </font><font style="color:rgb(26, 28, 30);">运行</font><font style="color:rgb(26, 28, 30);"> </font><font style="color:rgb(50, 48, 44);">JobManager</font><font style="color:rgb(26, 28, 30);"> </font><font style="color:rgb(26, 28, 30);">(任务调度、Checkpoint 协调)。同时运行着 Kafka/ZK/MySQL。</font>
+ **<font style="color:rgb(26, 28, 30);">Node 2 & Node 3 (Workers):</font>**<font style="color:rgb(26, 28, 30);"> 运行 </font><font style="color:rgb(50, 48, 44);">TaskManager</font><font style="color:rgb(26, 28, 30);"> (干活的)。</font>

<font style="color:rgb(26, 28, 30);">配置文件·：vim /opt/flink/conf/flink-conf.yaml</font>

```plain
# ===============================================================
# 基础配置
# ===============================================================
# [重要] JobManager 的地址，必须填 Node 1 的 hostname
jobmanager.rpc.address: node1

# JobManager 的端口 (默认 6123)
jobmanager.rpc.port: 6123

# JobManager 给多少内存？
# Node 1 还有 Kafka 和 MySQL，给 Flink JM 1GB 足够了
jobmanager.memory.process.size: 1024m

# ===============================================================
# TaskManager 配置 (Worker 节点)
# ===============================================================
# [重要] 每个 Worker 节点给多少内存？
# Node 2/3 总共 8G。
# 操作系统+缓存~2G，Storm(如果不跑)~0G，余量安全起见给 3G-4G。
# 这里我们给 3.2GB ，对齐storm ,留足余地，防止 OOM。
taskmanager.memory.process.size: 3276m

# [重要] 每个 TaskManager 提供多少个 Slot (并发槽位)？
# 你的机器是 4核。通常设置为 CPU 核心数或一半。
# 建议设为 2，留 2 个核处理网络中断、GC 和系统中断。
taskmanager.numberOfTaskSlots: 2

# 并行度默认值 (建议跟 Slot 总数一致：2台机器 * 2 Slot = 4)
parallelism.default: 4

# ===============================================================
# 实验相关配置 (At-Least-Once & Checkpoint)
# ===============================================================
# [重要] 状态后端。
# 因为你的磁盘是 ESSD Entry (较慢)，为了不让 IO 成为瓶颈，
# 我们直接把状态存在内存里 (HashMapStateBackend)。
state.backend: hashmap

# Checkpoint 存储路径 (必须配置，否则无法恢复)
# 这里用本地文件系统模拟 (生产环境通常是 HDFS/S3)
state.checkpoints.dir: file:///tmp/flink-checkpoints
state.savepoints.dir: file:///tmp/flink-savepoints

# 开启增量 Checkpoint (虽然 hashmap 主要是全量，但配上无妨)
state.backend.incremental: true

# ===============================================================
# Web UI
# ===============================================================
# 访问端口，默认 8081
rest.port: 8081
# 允许从任意 IP 访问 Dashboard
rest.bind-address: 0.0.0.0
```

storm.zookeeper.servers:
     - "node1"
#     - "server2"
# 
nimbus.seeds: ["node1"]

# ==========================================
# 本地状态存储路径 (不要放在默认的 /tmp 下，重启容易丢)
# ==========================================
storm.local.dir: "/opt/storm/data"

# ==========================================
# 资源限制 (针对 4核 8G 机器的优化)
# ==========================================
# [重要] 限制 mimbus 进程的最大堆内存为 1024 MB
# 建议将 Xms (初始堆) 和 Xmx (最大堆) 设置为相同的值，以避免内存抖动
nimbus.childopts: "-Xms1024m -Xmx1024m"

supervisor.childopts: "-Xms512m -Xmx512m"

# 默认是 2048MB (2G)，如果不改，启动 4 个 Worker 机器就挂了
# 每个 Worker 的 JVM Heap:1.6G
worker.heap.memory.mb: 1638
worker.childopts: "-Xms1638m -Xmx1638m"
topology.worker.max.heap.size.mb: 1638
# [重要] 定义每个节点可以跑多少个 Worker (Slots)
# 设置 2 个端口，代表最多可以跑 2 个并发。
# 对应你的 2 vCPU。
supervisor.slots.ports:
    - 6700
    - 6701

# ==========================================
# 容错实验相关设置
# ==========================================
# 消息超时时间。实验中你可能会杀节点，建议设大一点(60秒)
# 防止机器还没重启完，消息就因为超时被判失败了
topology.message.timeout.secs: 60

log.retention.hours=1 # 只保留1小时日志，省磁盘
